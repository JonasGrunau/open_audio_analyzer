Future<void> _loadFonts() async {
  Future<void> load(String family, List<String> paths) async {
    final loader = FontLoader(family);
    for (final path in paths) {
      final bytes = File(path).readAsBytesSync();
      loader.addFont(Future<ByteData>.value(ByteData.sublistView(bytes)));
    }
    await loader.load();
  }

  await load('Inter', [
    'assets/fonts/Inter-Regular.ttf',
    'assets/fonts/Inter-Medium.ttf',
    'assets/fonts/Inter-SemiBold.ttf',
  ]);
  await load('JetBrains Mono', [
    'assets/fonts/JetBrainsMono-Regular.ttf',
    'assets/fonts/JetBrainsMono-Medium.ttf',
  ]);
}

Future<void> _shoot(WidgetTester tester, String name) async {
  final boundary = tester.renderObject<RenderRepaintBoundary>(
    find.byKey(_root),
  );
  await tester.runAsync(() async {
    final image = await boundary.toImage(pixelRatio: 2);
    final data = await image.toByteData(format: ui.ImageByteFormat.png);
    Directory(_out).createSync(recursive: true);
    File('$_out/$_stage-$name.png').writeAsBytesSync(
      data!.buffer.asUint8List(),
    );
    image.dispose();
  });
}

/// A row of modules over one engine, at sizes the grid actually produces.
class _Rack extends StatefulWidget {
  const _Rack({
    required this.colors,
    required this.source,
    required this.tiles,
    this.pad = Space.md,
    this.cell = const Size(100, 70),
  });

  final OaaColors colors;
  final OaaSource source;

  /// `(kind, columns, rows)` — the grid cell is 100×70 here, close to a
  /// 1600×940 canvas divided the way the default preset divides it.
  final List<(ModuleKind, int, int)> tiles;

  /// Outer padding around the rack. Zero when a single module is captured for
  /// measurement, so the image *is* the module.
  final double pad;

  /// Grid cell size, so a capture can use the one a real canvas produces.
  final Size cell;

  @override
  State<_Rack> createState() => _RackState();
}

class _RackState extends State<_Rack> with SingleTickerProviderStateMixin {
  late final OaaEngine engine = OaaEngine.start(source: widget.source);
  late final MeterClock clock = MeterClock(engine: engine, vsync: this);

  @override
  void dispose() {
    clock.dispose();
    engine.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) => OaaTheme(
    colors: widget.colors,
    child: Material(
      color: widget.colors.background,
      child: Padding(
        padding: EdgeInsets.all(widget.pad),
        child: Row(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            for (final (index, tile) in widget.tiles.indexed) ...[
              if (index > 0) const SizedBox(width: Space.md),
              SizedBox(
                width: tile.$2 * widget.cell.width,
                height: tile.$3 * widget.cell.height,
                child: ModuleHost(
                  spec: ModuleSpec(
                    id: 'm$index',
                    kind: tile.$1,
                    rect: GridRect(
                      column: 0,
                      row: 0,
                      columns: tile.$2,
                      rows: tile.$3,
                    ),
                  ),
                  engine: engine,
                  clock: clock,
                  calibration: BuiltInCalibrations.fallback,
                  selected: false,
                  onMenu: () {},
                ),
              ),
            ],
          ],
        ),
      ),
    ),
  );
}

Future<void> _pumpRack(
  WidgetTester tester, {
  required OaaColors colors,
  required List<(ModuleKind, int, int)> tiles,
  OaaSource source = OaaSource.testTone,
  double pad = Space.md,
  Size cell = const Size(100, 70),
  required Size size,
}) async {
  tester.view.physicalSize = size * 2;
  tester.view.devicePixelRatio = 2;
  addTearDown(tester.view.reset);

  final container = ProviderContainer();
  addTearDown(container.dispose);

  await tester.pumpWidget(
    UncontrolledProviderScope(
      container: container,
      child: MaterialApp(
        debugShowCheckedModeBanner: false,
        home: RepaintBoundary(
          key: _root,
          child: _Rack(
            colors: colors,
            source: source,
            tiles: tiles,
            pad: pad,
            cell: cell,
          ),
        ),
      ),
    ),
  );
  await tester.pump();
  await tester.pump(const Duration(milliseconds: 600));
}

// --- The settings panel, opened the way the application opens it ------------

class _Opener extends StatelessWidget {
  const _Opener();

  @override
  Widget build(BuildContext context) => Material(
    child: Center(
      child: GestureDetector(
        onTap: () => showOaaPanel<void>(
          context: context,
          builder: (context) => const SettingsPanel(),
        ),
        child: const Text('open'),
      ),
    ),
  );
}

Future<void> _pumpSettings(WidgetTester tester, OaaColors colors) async {
  tester.view.physicalSize = const Size(1200, 900) * 2;
  tester.view.devicePixelRatio = 2;
  addTearDown(tester.view.reset);

  final store = ConfigStore.disabled();
  addTearDown(store.dispose);
  final container = ProviderContainer(
    overrides: [
      configStoreProvider.overrideWithValue(store),
      startupConfigProvider.overrideWithValue(const StartupConfig()),
    ],
  );
  addTearDown(container.dispose);

  await tester.pumpWidget(
    UncontrolledProviderScope(
      container: container,
      child: RepaintBoundary(
        key: _root,
        child: MaterialApp(
          debugShowCheckedModeBanner: false,
          theme: oaaThemeData(colors),
          home: OaaTheme(colors: colors, child: const _Opener()),
        ),
      ),
    ),
  );
  await tester.tap(find.text('open'));
  await tester.pumpAndSettle();
}

// --- The whole application, for the status bar -------------------------------

Future<void> _pumpApp(WidgetTester tester) async {
  tester.view.physicalSize = const Size(1600, 940) * 2;
  tester.view.devicePixelRatio = 2;
  addTearDown(tester.view.reset);

  final store = ConfigStore.disabled();
  addTearDown(store.dispose);
  final container = ProviderContainer(
    overrides: [
      configStoreProvider.overrideWithValue(store),
      startupConfigProvider.overrideWithValue(const StartupConfig()),
    ],
  );
  addTearDown(container.dispose);

  await tester.pumpWidget(
    UncontrolledProviderScope(
      container: container,
      child: RepaintBoundary(key: _root, child: const OaaApp()),
    ),
  );
  await tester.pump();
  await tester.pump(const Duration(milliseconds: 600));
}

void main() {
  setUpAll(_loadFonts);

  final dark = OaaColors.precisionInstrument;
  final light = oaaColorsFromSkin(BuiltInSkins.daylight);

  testWidgets('vu — dark, silence (needle at rest)', (tester) async {
    await _pumpRack(
      tester,
      colors: dark,
      source: OaaSource.silence,
      tiles: const [(ModuleKind.vuMeter, 6, 5)],
      size: const Size(640, 380),
    );
    await _shoot(tester, 'vu-dark-rest');
  });

  testWidgets('vu — dark, tone (needle deflected)', (tester) async {
    await _pumpRack(
      tester,
      colors: dark,
      tiles: const [(ModuleKind.vuMeter, 6, 5)],
      size: const Size(640, 380),
    );
    await _shoot(tester, 'vu-dark-tone');
  });

  testWidgets('vu — dark, wide and short', (tester) async {
    await _pumpRack(
      tester,
      colors: dark,
      tiles: const [(ModuleKind.vuMeter, 9, 4)],
      size: const Size(940, 310),
    );
    await _shoot(tester, 'vu-dark-wide');
  });

  testWidgets('vu — light', (tester) async {
    await _pumpRack(
      tester,
      colors: light,
      tiles: const [(ModuleKind.vuMeter, 6, 5)],
      size: const Size(640, 380),
    );
    await _shoot(tester, 'vu-light');
  });

  testWidgets('tracks — dark', (tester) async {
    await _pumpRack(
      tester,
      colors: dark,
      tiles: const [
        (ModuleKind.superMeter, 8, 8),
        (ModuleKind.lufsMeter, 5, 8),
        (ModuleKind.digitalMeter, 4, 8),
      ],
      size: const Size(1780, 610),
    );
    await _shoot(tester, 'tracks-dark');
  });

  testWidgets('tracks — light', (tester) async {
    await _pumpRack(
      tester,
      colors: light,
      tiles: const [
        (ModuleKind.superMeter, 8, 8),
        (ModuleKind.lufsMeter, 5, 8),
        (ModuleKind.digitalMeter, 4, 8),
      ],
      size: const Size(1780, 610),
    );
    await _shoot(tester, 'tracks-light');
  });

  // Every module at its default size, so padding and centring can be compared
  // across the twelve rather than one at a time.
  const sheetA = <(ModuleKind, int, int)>[
    (ModuleKind.superMeter, 8, 8),
    (ModuleKind.lufsMeter, 5, 8),
    (ModuleKind.digitalMeter, 4, 8),
    (ModuleKind.vuMeter, 6, 5),
  ];
  const sheetB = <(ModuleKind, int, int)>[
    (ModuleKind.phaseScope, 6, 6),
    (ModuleKind.stereoCloud, 6, 6),
    (ModuleKind.validator, 6, 4),
    (ModuleKind.alertMeter, 4, 3),
    (ModuleKind.numberBox, 4, 2),
  ];
  const sheetC = <(ModuleKind, int, int)>[
    (ModuleKind.spectrumAnalyzer, 12, 7),
    (ModuleKind.spectrogram, 12, 7),
    (ModuleKind.histogram, 12, 6),
  ];

  for (final (name, tiles, size) in <(String, List<(ModuleKind, int, int)>, Size)>[
    ('a', sheetA, Size(2400, 610)),
    ('b', sheetB, Size(2720, 470)),
    ('c', sheetC, Size(3680, 530)),
  ]) {
    testWidgets('sheet $name — dark', (tester) async {
      await _pumpRack(tester, colors: dark, tiles: tiles, size: size);
      await _shoot(tester, 'sheet-$name-dark');
    });

    testWidgets('sheet $name — light', (tester) async {
      await _pumpRack(tester, colors: light, tiles: tiles, size: size);
      await _shoot(tester, 'sheet-$name-light');
    });
  }

  const solo = <(ModuleKind, int, int)>[
    (ModuleKind.superMeter, 8, 8),
    (ModuleKind.lufsMeter, 5, 8),
    (ModuleKind.digitalMeter, 4, 8),
    (ModuleKind.vuMeter, 6, 5),
    (ModuleKind.phaseScope, 6, 6),
    (ModuleKind.stereoCloud, 6, 6),
    (ModuleKind.validator, 6, 4),
    (ModuleKind.alertMeter, 4, 3),
    (ModuleKind.numberBox, 4, 2),
    (ModuleKind.spectrumAnalyzer, 12, 7),
    (ModuleKind.spectrogram, 12, 7),
    (ModuleKind.histogram, 12, 6),
  ];

  for (final tile in solo) {
    testWidgets('solo ${tile.$1.name}', (tester) async {
      await _pumpRack(
        tester,
        colors: dark,
        tiles: [tile],
        pad: 0,
        size: Size(tile.$2 * 100, tile.$3 * 70),
      );
      await _shoot(tester, 'solo-${tile.$1.name}');
    });
  }

  // Every kind at its *minimum* legal size, on the cell size a 1600x880
  // canvas produces. A module that draws nothing here is worse than one that
  // is badly padded.
  const minima = <(ModuleKind, int, int)>[
    (ModuleKind.numberBox, 2, 1),
    (ModuleKind.numberBox, 2, 2),
    (ModuleKind.alertMeter, 3, 2),
    (ModuleKind.validator, 4, 3),
    (ModuleKind.vuMeter, 5, 4),
    (ModuleKind.digitalMeter, 3, 6),
    (ModuleKind.lufsMeter, 4, 6),
    (ModuleKind.superMeter, 6, 6),
    (ModuleKind.phaseScope, 5, 5),
    (ModuleKind.stereoCloud, 6, 5),
    (ModuleKind.histogram, 8, 5),
    (ModuleKind.spectrumAnalyzer, 8, 5),
  ];

  for (final tile in minima) {
    testWidgets('min ${tile.$1.name} ${tile.$2}x${tile.$3}', (tester) async {
      await _pumpRack(
        tester,
        colors: dark,
        tiles: [tile],
        pad: 0,
        // 1600 wide / 24 columns, 880 tall / 16 rows, less an 8 px gutter.
        cell: const Size(66.7, 55),
        size: Size(tile.$2 * 66.7, tile.$3 * 55.0),
      );
      await _shoot(tester, 'min-${tile.$1.name}-${tile.$2}x${tile.$3}');
    });
  }

  testWidgets('settings — dark', (tester) async {
    await _pumpSettings(tester, dark);
    await _shoot(tester, 'settings-dark');
  });

  testWidgets('app — dark, the status bar', (tester) async {
    await _pumpApp(tester);
    await _shoot(tester, 'app-dark');
  });
}
